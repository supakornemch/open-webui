🐍 Python — Create Chat + Completion

# Create a chat session, then get AI reply
# 1. Create chat with user + assistant placeholder
import requests, uuid

BASE = "https://genie.haadthip.com"
HEADERS = {"Authorization": "Bearer YOUR_KEY", "Content-Type": "application/json"}

user_id = str(uuid.uuid4())
asst_id = str(uuid.uuid4())

# Create chat
chat = requests.post(f"{BASE}/api/v1/chats/new", headers=HEADERS, json={
  "chat": {
    "models": ["gpt-5.4-mini"],
    "messages": [
      {"id": user_id, "role": "user",
       "content": "What is MFA?", "childrenIds": [asst_id]},
      {"id": asst_id, "role": "assistant",
       "content": "", "parentId": user_id, "childrenIds": [],
       "model": "gpt-5.4-mini", "done": False}
    ],
    "history": {
      "currentId": asst_id,
      "messages": {
        user_id: {"id": user_id, "role": "user",
                  "content": "What is MFA?", "childrenIds": [asst_id]},
        asst_id: {"id": asst_id, "role": "assistant",
                  "content": "", "parentId": user_id,
                  "childrenIds": [], "model": "gpt-5.4-mini", "done": False}
      }
    }
  }
}).json()
chat_id = chat["id"]

# 2. Trigger AI completion
response = requests.post(f"{BASE}/api/chat/completions", headers=HEADERS, json={
  "chat_id": chat_id,
  "id": asst_id,
  "model": "gpt-5.4-mini",
  "messages": [{"role": "user", "content": "What is MFA?"}],
  "stream": False
})
print(response.json())

More PoC — MCP Proxy

Use MCP (Model Context Protocol) to connect AI to internal databases

💡

The Idea

Deploy an MCP Proxy Server on Azure App Service that sits between Open WebUI and internal databases (PostgreSQL, MSSQL). This lets users ask questions in natural language and get answers directly from company databases.

🐍 Python FastMCP — Minimal Example

# requirements.txt
# fastmcp>=0.1.0 uvicorn>=0.22.0 psycopg2-binary>=2.9.0 pyodbc>=5.0.0 azure-identity>=1.15.0

# main.py
from fastmcp import FastMCP
import psycopg2

mcp = FastMCP("DB Proxy", host="0.0.0.0", port=8000)

@mcp.tool()
def query_postgres(sql: str) -> str:
    """Run a read-only SQL query on PostgreSQL"""
    conn = psycopg2.connect("postgresql://user:pass@host/db")
    cur = conn.cursor()
    cur.execute(sql)
    rows = cur.fetchall()
    conn.close()
    return "\n".join(str(r) for r in rows)

@mcp.tool()
def query_mssql(sql: str) -> str:
    """Run a read-only SQL query on MSSQL"""
    import pyodbc
    conn = pyodbc.connect("DRIVER={ODBC Driver 18};SERVER=host;DB=db;UID=user;PWD=pass")
    cur = conn.cursor()
    cur.execute(sql)
    rows = cur.fetchall()
    conn.close()
    return "\n".join(str(r) for r in rows)

# Deploy to Azure App Service, then register in Open WebUI as MCP Server
🔌

Register in Open WebUI

  1. Go to Admin Panel → Settings → Connections
  2. Add new MCP Server
  3. Set URL to https://your-app.azurewebsites.net/mcp
  4. Choose SSE transport type
  5. Save → Chat can now query databases via MCP tools!
